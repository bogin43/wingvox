# ---------- macOS ----------

# mlx-whisper itself is NOT listed here: install.sh installs it separately
# with --no-deps, because its declared dependency set drags in torch
# (106MB), llvmlite (39MB), scipy (19MB), numba, sympy and networkx --
# ~176MB that stt_mac.py's code path never touches. torch is never imported
# at all, and numba/scipy are only reached by mlx_whisper's unused
# word-timestamps feature, which flow.py fakes out with sys.modules stubs
# (see the IS_ARM_MAC block near the top of flow.py). The installer used to
# download all of it and then uninstall it again on every single install.
#
# The lines below are mlx-whisper's remaining, genuinely-needed
# dependencies, spelled out because --no-deps means pip won't resolve them.
# If a future mlx-whisper release imports something new, it fails loudly
# with an ImportError rather than silently degrading.
#
# Installing by hand? Use:
#   pip install -r requirements.txt && pip install --no-deps mlx-whisper
#
# mlx has no x86_64 build at all -- Apple-Silicon architecture only, not a
# version gap -- so this tier is gated on platform_machine == "arm64" too, or
# pip would try (and fail) to resolve mlx on an Intel Mac.
mlx; sys_platform == "darwin" and platform_machine == "arm64"
more-itertools; sys_platform == "darwin" and platform_machine == "arm64"
tiktoken; sys_platform == "darwin" and platform_machine == "arm64"
tqdm; sys_platform == "darwin" and platform_machine == "arm64"
huggingface_hub; sys_platform == "darwin" and platform_machine == "arm64"

# Intel Macs use the same faster-whisper/CTranslate2 backend as Windows (see
# stt_intel_mac.py, a thin re-export of stt_windows.py). ctranslate2
# publishes native macosx_11_0_x86_64 wheels, confirmed via PyPI.
faster-whisper; sys_platform == "darwin" and platform_machine == "x86_64"

# Needed to build Wingvox.app (setup.py's py2app step). Installed explicitly
# here rather than relying on setuptools' `setup_requires` auto-fetch, which
# uses a deprecated "egg" mechanism that fails to pull in py2app's own
# dependencies (modulegraph, macholib) correctly in this environment.
py2app; sys_platform == "darwin"

# verified working in development.

pyobjc-framework-Cocoa==12.2.1; sys_platform == "darwin"
pyobjc-framework-Quartz==12.2.1; sys_platform == "darwin"
pyobjc-framework-ApplicationServices==12.2.1; sys_platform == "darwin"
